    왜 클로드일까

    일을 맡기는 AI, 그리고 오늘의 질문 하나 — “이거, 매주 똑같이 하고 있나?”

    방식 강사 이야기준비 없음보기만
    왜 클로드로 하나요?

    AI는 지금 묻고 답하는 AI에서 일을 맡기는 AI(에이전트 AI)로 넘어가고 있습니다. 챗GPT·제미나이·클로드 모두 그쪽으로 가고 있습니다.

    그중 일을 맡기는 쪽은 지금 클로드가 가장 앞서 있다고 봅니다. 여러 AI를 두루 써 보고 오늘 클로드를 고른 이유입니다.

    가르친 방법도 달랐습니다. 사람이 답마다 점수를 매겨 길들이는 대신, 지켜야 할 원칙을 글로 적어 두고(‘헌법’이라고 부릅니다) 그 원칙에 비추어 스스로 답을 고치도록 훈련했습니다. 오늘 여러 번 보실 “지어내지 않고 되묻는” 모습이 이 방식과 닿아 있습니다.

    그래도 오늘 배우시는 시키는 방법은 어느 AI에서나 똑같이 통합니다. 챗GPT를 쓰시는 분도 그대로 가져가시면 됩니다.

    /**
 * 오늘의 커피 담당자 — 구글 시트 연결용 앱스 스크립트
 * 룰렛 돌리기 · 새 접수 · 남의 주문 지우기는 아래 강사 암호가 있어야 됩니다.
 * 룰렛은 한 번에 1~5명까지 뽑을 수 있습니다.
 */
var HOST_KEY = '강사암호';   // ← 따옴표 안에 강사 암호를 넣으세요

var SS = SpreadsheetApp.getActiveSpreadsheet();

function sheet_(name, headers) {
  var s = SS.getSheetByName(name);
  if (!s) { s = SS.insertSheet(name); s.appendRow(headers); }
  return s;
}
function readState_() {
  var s = sheet_('상태', ['키', '값']);
  var v = s.getDataRange().getValues().slice(1), o = {};
  v.forEach(function (r) { if (r[0] !== '') o[r[0]] = r[1]; });
  if (!o.round) o.round = 1;
  return o;
}
function writeState_(o) {
  var s = sheet_('상태', ['키', '값']);
  s.clearContents();
  s.appendRow(['키', '값']);
  Object.keys(o).forEach(function (k) { s.appendRow([k, o[k]]); });
}
function ordersSheet_() { return sheet_('주문', ['회차', '번호', '이름', '메뉴', '등록 시각', '확인값']); }
function orders_(round) {
  var v = ordersSheet_().getDataRange().getValues().slice(1);
  return v.filter(function (r) { return String(r[0]) === String(round); })
          .map(function (r) { return { id: String(r[1]), name: String(r[2]), menu: String(r[3]), token: String(r[5]) }; });
}
function out_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
function view_() {
  var st = readState_();
  var list = orders_(st.round).map(function (o) { return { id: o.id, name: o.name, menu: o.menu }; });
  var ws = [];
  if (st.winners) { try { ws = JSON.parse(st.winners); } catch (e) { ws = []; } }
  else if (st.winnerId) { ws = [{ id: String(st.winnerId), name: String(st.winnerName), menu: String(st.winnerMenu) }]; }
  var w = ws.length ? { id: ws[0].id, name: ws[0].name, menu: ws[0].menu, at: String(st.spinAt) } : null;
  return { ok: true, round: Number(st.round), orders: list, winner: w, winners: ws };
}
function clean_(t, n) { return String(t || '').replace(/[\r\n\t]/g, ' ').trim().slice(0, n); }

function doGet() { return out_(view_()); }

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var b = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var st = readState_();
    var isHost = String(b.key || '') === HOST_KEY;

    if (b.action === 'add') {
      var name = clean_(b.name, 12), menu = clean_(b.menu, 24);
      if (!name || !menu) return out_({ ok: false, error: '이름과 메뉴를 모두 적어 주세요.' });
      if (orders_(st.round).length >= 80) return out_({ ok: false, error: '이번 회차는 가득 찼습니다.' });
      var id = Utilities.getUuid().slice(0, 8);
      ordersSheet_().appendRow([st.round, id, name, menu, new Date(), clean_(b.token, 40)]);
    }
    else if (b.action === 'delete') {
      var sh = ordersSheet_(), v = sh.getDataRange().getValues();
      for (var i = v.length - 1; i >= 1; i--) {
        if (String(v[i][0]) === String(st.round) && String(v[i][1]) === String(b.id)) {
          if (isHost || (b.token && String(v[i][5]) === String(b.token))) sh.deleteRow(i + 1);
          else return out_({ ok: false, error: '내가 등록한 주문만 지울 수 있습니다.' });
          break;
        }
      }
    }
    else if (b.action === 'spin') {
      if (!isHost) return out_({ ok: false, error: '룰렛은 강사만 돌릴 수 있습니다.' });
      var list = orders_(st.round);
      if (list.length < 2) return out_({ ok: false, error: '두 분 이상 등록해야 돌릴 수 있습니다.' });
      var cnt = Math.max(1, Math.min(5, Number(b.count) || 1, list.length));
      var pool = list.slice();
      for (var i = pool.length - 1; i > 0; i--) { var j = Math.floor(Math.random() * (i + 1)); var t = pool[i]; pool[i] = pool[j]; pool[j] = t; }
      var picked = pool.slice(0, cnt).map(function (o) { return { id: o.id, name: o.name, menu: o.menu }; });
      writeState_({ round: st.round, winners: JSON.stringify(picked), spinAt: String(Date.now()) });
    }
    else if (b.action === 'reset') {
      if (!isHost) return out_({ ok: false, error: '새 접수는 강사만 시작할 수 있습니다.' });
      writeState_({ round: Number(st.round) + 1 });
    }
    return out_(view_());
  } finally {
    lock.releaseLock();
  }
}
